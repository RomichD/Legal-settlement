async function calcGK395() { const debt = parseFloat(document.getElementById('gk395-debt').value); const startStr = document.getElementById('gk395-start').value; const endStr = document.getElementById('gk395-end').value; const region = document.getElementById('gk395-region').value; const resultDiv = document.getElementById('gk395-result'); if (isNaN(debt) || debt <= 0) { resultDiv.className = 'result-box error'; resultDiv.textContent = 'Введите корректную сумму задолженности.'; return; } if (!startStr || !endStr) { resultDiv.className = 'result-box error'; resultDiv.textContent = 'Заполните даты начала и окончания просрочки.'; return; } const start = parseDate(startStr); const end = parseDate(endStr); if (end <= start) { resultDiv.className = 'result-box error'; resultDiv.textContent = 'Дата окончания должна быть позже даты начала.'; return; } const rows = document.querySelectorAll('#gk395-payment-tbody tr'); const payments = []; rows.forEach(tr => { const dateVal = tr.querySelector('.gk395-pay-date').value; const sumVal = tr.querySelector('.gk395-pay-sum').value; if (dateVal && sumVal) { const dt = parseDate(dateVal); if (dt > start && dt < end) { payments.push({ date: dt, sum: parseFloat(sumVal) }); } } }); payments.sort((a, b) => a.date - b.date); await updateKeyRate(); function getRateForDate(date) { const dateObj = new Date(date); if (dateObj < new Date(2015, 5, 1)) { return 8.25; } if (dateObj >= new Date(2015, 5, 1) && dateObj < new Date(2016, 7, 1)) { const r = getRegionRateForDate(date, region); if (r !== null) return r; } const keyRate = getKeyRateForDate(date); if (keyRate !== null) return keyRate; return DEFAULT_KEY_RATE; } let totalInterest = 0; let details = []; let currentDebt = debt; let currentDate = new Date(start); let paymentIndex = 0; while (currentDate < end) { let nextDate; if (paymentIndex < payments.length && payments[paymentIndex].date > currentDate) { nextDate = new Date(payments[paymentIndex].date); } else { nextDate = new Date(end); } const days = daysBetween(currentDate, nextDate); if (days > 0) { const rate = getRateForDate(currentDate); const interest = currentDebt * (rate / 100) * days / 365; totalInterest += interest; details.push( `Период с ${formatDate(currentDate)} по ${formatDate(nextDate)}: ` + `долг ${currentDebt.toFixed(2)} руб., ставка ${rate.toFixed(2)}%, ` + `${days} дн. → ${interest.toFixed(2)} руб.` ); } if (paymentIndex < payments.length && payments[paymentIndex].date <= nextDate) { const paySum = payments[paymentIndex].sum; if (paySum > currentDebt) { details.push(`⚠️ Платёж ${formatDate(payments[paymentIndex].date)} на ${paySum.toFixed(2)} руб. превышает остаток долга (${currentDebt.toFixed(2)} руб.). Учитываем только ${currentDebt.toFixed(2)} руб.`); currentDebt = 0; } else { currentDebt -= paySum; details.push(`Платёж ${formatDate(payments[paymentIndex].date)} на ${paySum.toFixed(2)} руб. Остаток долга: ${currentDebt.toFixed(2)} руб.`); } paymentIndex++; } currentDate = new Date(nextDate); if (currentDebt < 0.01) { details.push('✅ Долг полностью погашен.'); break; } } const output = `Сумма задолженности: ${debt.toFixed(2)} руб.\n` + `Период: с ${formatDate(start)} по ${formatDate(end)}\n` + `Всего процентов по ст. 395 ГК РФ: ${totalInterest.toFixed(2)} руб.\n\n` + `Детали расчёта:\n${details.join('\n')}`; resultDiv.className = 'result-box'; resultDiv.textContent = output; }
